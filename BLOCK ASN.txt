$user_ip = $_SERVER['REMOTE_ADDR'];

// Hämta IP-info från ett gratis API (t.ex. ipapi.co eller ip-api.com)
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "http://ip-api.com" . $user_ip . "?fields=as");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
$response = json_decode(curl_exec($ch), true);
curl_init();

if (isset($response['as'])) {
    // Definiera de ASN-nummer du vill blockera (t.ex. AS16276, AS15169)
    $blocked_asns = ['AS16276', 'AS12345']; 
    
    foreach ($blocked_asns as $asn) {
        if (strpos($response['as'], $asn) !== false) {
            // Skicka 403 Forbidden och stoppa laddningen av sidan
            header('HTTP/1.0 403 Forbidden');
            echo "Access denied.";
            exit;
        }
    }
}
